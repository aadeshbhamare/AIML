# Create a class named PersonalAssistantAgent
class PersonalAssistantAgent:

    # Constructor method
    # It is automatically executed when the object is created
    def __init__(self):

        # Display the title of the practical
        print("\n========== PERSONAL ASSISTANT AGENT ==========\n")

        # Display PEAS Representation
        print("PEAS Representation")
        print("------------------------------")

        # Display Performance Measure
        print("Performance Measure : Accurate assistance, quick response and user satisfaction.")

        # Display Environment
        print("Environment         : User, computer and personal assistant interface.")

        # Display Actuators
        print("Actuators           : Displaying information and reminder messages.")

        # Display Sensors
        print("Sensors             : User commands through keyboard.")

        # Display separator
        print("------------------------------\n")

    # Function to process user commands and generate responses
    def respond(self, command):

        # Display that the agent is processing the command
        print("\nProcessing user command...")

        # Convert command into lowercase
        command = command.lower()

        # ---------------- DECISION MAKING ----------------

        # Check whether the user asks for help
        if "help" in command:
            response = "I can help you with time, reminders and daily tasks."

        # Check whether the user asks to set a reminder
        elif "reminder" in command:
            task = input("Enter your reminder task: ")
            response = "Reminder saved for: " + task

        # Check whether the user asks about the time
        elif "time" in command:
            from datetime import datetime
            response = "Current time is: " + datetime.now().strftime("%H:%M:%S")

        # Check whether the user asks for daily task suggestions
        elif "task" in command:
            response = "You should prepare your schedule and complete important tasks first."

        # Check whether the user greets the assistant
        elif "hello" in command or "hi" in command:
            response = "Hello! How can I assist you today?"

        # Default response for unknown commands
        else:
            response = "Sorry, I could not understand your command."

        # Return the response generated by the agent
        return response


# ---------------------------------------------------------
# MAIN PROGRAM STARTS HERE
# ---------------------------------------------------------

# Create an object of PersonalAssistantAgent class
agent = PersonalAssistantAgent()

# Display the working steps of the intelligent agent
print("Agent Behaviour")
print("------------------------------")
print("1. Sense  -> Receive user command")
print("2. Think  -> Understand and process the command")
print("3. Act    -> Perform the required action")
print("4. Repeat -> Continue assisting the user")
print("------------------------------")

# Start an infinite loop
while True:

    # ---------------- SENSOR ----------------

    # Read command from the user
    user_input = input("\nYou: ")

    # Check whether the user wants to exit
    if user_input.lower() == "exit":

        # Display successful completion message
        print("\nPersonal Assistant executed successfully!")

        # Display thank-you message
        print("Thank you!")

        # Exit the loop
        break

    # ---------------- THINK ----------------

    # Pass user command to the intelligent agent
    response = agent.respond(user_input)

    # ---------------- ACTUATOR ----------------

    # Display the response generated by the agent
    print("\nAssistant:", response)

    # Display separator
    print("-----------------------------------")
