# Create a class named ChatbotAgent
class ChatbotAgent:

    # Constructor method
    # It is automatically executed when the object is created
    def __init__(self):

        # Display the title of the practical
        print("\n========== CHATBOT INTELLIGENT AGENT ==========\n")

        # Display PEAS Representation
        print("PEAS Representation")
        print("------------------------------")

        # Display Performance Measure
        print("Performance Measure : Accurate responses, user satisfaction and fast response.")

        # Display Environment
        print("Environment         : User and chatbot interface.")

        # Display Actuators
        print("Actuators           : Displaying text responses and messages.")

        # Display Sensors
        print("Sensors             : User messages through keyboard.")

        # Display separator
        print("------------------------------\n")

    # Function to process user messages and generate responses
    def respond(self, message):

        # Display that the agent is processing the message
        print("\nProcessing user message...")

        # Convert message into lowercase
        message = message.lower()

        # ---------------- RESPONSE DECISION ----------------

        # Check whether the user greets the chatbot
        if "hello" in message or "hi" in message:
            response = "Hello! How can I help you?"

        # Check whether the user asks about the chatbot
        elif "your name" in message:
            response = "I am an Intelligent Chatbot."

        # Check whether the user asks how it is
        elif "how are you" in message:
            response = "I am fine! Thank you for asking."

        # Check whether the user asks about Artificial Intelligence
        elif "artificial intelligence" in message or "what is ai" in message:
            response = "Artificial Intelligence enables machines to perform tasks that require human intelligence."

        # Check whether the user asks about Machine Learning
        elif "machine learning" in message:
            response = "Machine Learning allows machines to learn from data and make predictions."

        # Check whether the user says thank you
        elif "thank you" in message or "thanks" in message:
            response = "You are welcome! Happy to help."

        # Default response for unknown messages
        else:
            response = "Sorry, I did not understand. Please ask another question."

        # Return the response generated by the chatbot
        return response


# ---------------------------------------------------------
# MAIN PROGRAM STARTS HERE
# ---------------------------------------------------------

# Create an object of ChatbotAgent class
agent = ChatbotAgent()

# Display the working steps of the intelligent agent
print("Agent Behaviour")
print("------------------------------")
print("1. Sense  -> Receive user message")
print("2. Think  -> Understand the message")
print("3. Act    -> Generate a suitable response")
print("4. Repeat -> Continue the conversation")
print("------------------------------")

# Start an infinite loop
while True:

    # ---------------- SENSOR ----------------

    # Read message from the user
    user_input = input("\nYou: ")

    # Check whether the user wants to exit
    if user_input.lower() == "bye":

        # Display goodbye message
        print("\nChatbot: Goodbye! Have a nice day!")

        # Display successful completion message
        print("\nChatbot Agent executed successfully!")

        # Display thank-you message
        print("Thank you!")

        # Exit the loop
        break

    # ---------------- THINK ----------------

    # Pass user message to the intelligent agent
    response = agent.respond(user_input)

    # ---------------- ACTUATOR ----------------

    # Display the response generated by the chatbot
    print("\nChatbot:", response)

    # Display separator
    print("-----------------------------------")
